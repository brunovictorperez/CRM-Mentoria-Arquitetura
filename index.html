Conecte o app para gravar e ler direto do Google Sheets em tempo real

Por que configurar? Links públicos de CSV do Google Sheets são apenas para leitura. Para que o aplicativo envie novos leads e edições de volta para a planilha instantaneamente, usamos um pequeno script gratuito na sua própria planilha.

Verificação Automática

Atualiza sozinho a cada 20 segundos em segundo plano

Passo a passo (leva 1 minuto)
  1. Abra sua planilha do Google Sheets e clique no menu Extensões > Apps Script.
  2. Apague qualquer código que estiver lá e cole o código abaixo:
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  const rows = sheet.getDataRange().getDisplayValues();
  const headers = rows[0].map(h => h.trim());
  const data = rows.slice(1).map((r, idx) => {
    let obj = { _rowIndex: idx + 2 };
    headers.forEach((h, i) => obj[h] = r[i] || "");
    return obj;
  });

  // Lê também as opções do menu suspenso (Validação de Dados) da coluna "Fase"
  let phaseOptions = [];
  const faseColIdx = headers.indexOf("Fase") + 1;
  if (faseColIdx > 0 && sheet.getLastRow() >= 2) {
    const rule = sheet.getRange(2, faseColIdx).getDataValidation();
    if (rule && rule.getCriteriaType() === SpreadsheetApp.DataValidationCriteria.VALUE_IN_LIST) {
      phaseOptions = rule.getCriteriaValues()[0] || [];
    }
  }

  return ContentService.createTextOutput(JSON.stringify({ status: "ok", data, phaseOptions }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0].map(h => h.trim());
  const rowValues = headers.map(h => payload.lead[h] !== undefined ? payload.lead[h] : "");

  if (payload.action === "add") {
    sheet.appendRow(rowValues);
  } else if (payload.action === "edit" && payload.rowIndex) {
    sheet.getRange(Number(payload.rowIndex), 1, 1, headers.length).setValues([rowValues]);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}

  3. Em "Quem pode acessar", selecione Qualquer pessoa e clique em Implantar.
  4. Copie a URL do App da Web gerada e cole no campo acima!
