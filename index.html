            <pre id="gasCodeBlock" class="bg-neutral-900 text-neutral-100 p-3 rounded-xl text-[10px] overflow-x-auto font-mono leading-normal">function doGet(e) {

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];

  const rows = sheet.getDataRange().getDisplayValues();

  const headers = rows[0].map(h => h.trim());

  const data = rows.slice(1).map((r, idx) => {

    let obj = { _rowIndex: idx + 2 };

    headers.forEach((h, i) => obj[h] = r[i] || "");

    return obj;

  });


  // Lê também as opções do menu suspenso (Validação de Dados) da coluna "Fase"

  let phaseOptions = [];

  const faseColIdx = headers.indexOf("Fase") + 1;

  if (faseColIdx > 0 && sheet.getLastRow() >= 2) {

    const rule = sheet.getRange(2, faseColIdx).getDataValidation();

    if (rule && rule.getCriteriaType() === SpreadsheetApp.DataValidationCriteria.VALUE_IN_LIST) {

      phaseOptions = rule.getCriteriaValues()[0] || [];

    }

  }


  return ContentService.createTextOutput(JSON.stringify({ status: "ok", data, phaseOptions }))

    .setMimeType(ContentService.MimeType.JSON);

}


function doPost(e) {

  const payload = JSON.parse(e.postData.contents);

  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];

  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0].map(h => h.trim());

  const rowValues = headers.map(h => payload.lead[h] !== undefined ? payload.lead[h] : "");


  if (payload.action === "add") {

    sheet.appendRow(rowValues);

  } else if (payload.action === "edit" && payload.rowIndex) {

    sheet.getRange(Number(payload.rowIndex), 1, 1, headers.length).setValues([rowValues]);

  }

  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))

    .setMimeType(ContentService.MimeType.JSON);

}</pre>

    async function fetchCsv(url) {

      const timestampUrl = `${url}&_t=${Date.now()}`;

      const response = await fetch(timestampUrl, { cache: "no-store" });

      const csvText = await response.text();

      return new Promise((resolve) => {

        Papa.parse(csvText, {

          header: true,

          skipEmptyLines: true,

          complete: (results) => {

            // Adiciona _rowIndex baseado na linha da planilha (Cabeçalho = 1, primeiro item = 2)

            const indexed = results.data.map((row, idx) => ({

              ...row,

              _rowIndex: idx + 2

            }));

            resolve(indexed);

          }

        });

      });

    }

    // Separa o que está entre parênteses (ex: "(O aluno vai relatar a dificuldade dele)") da área copiável

    function extractCopyableAndNotes(rawText) {

      const notes = [];

      const regex = /\(([^()]+)\)/g;

      let match;

      while ((match = regex.exec(rawText)) !== null) {

        notes.push(match[0]);

      }


      // Remove os trechos entre parênteses do texto que será copiado e limpa espaços/linhas extras

      const copyableText = rawText

        .replace(/\s*\([^()]+\)/g, '')

        .replace(/[ \t]+\n/g, '\n')

        .replace(/\n{3,}/g, '\n\n')

        .trim();


      return {

        rawText: rawText.trim(),

        copyableText,

        notes

      };

    }


    function parseApproaches(rawCsv) {

      const lines = rawCsv.split(/\r?\n/).map(l => l.replace(/^"|"$/g, '').trim()).filter(l => l.length > 0);

      const blocks = [];

      let currentBlock = null;


      lines.forEach(line => {

        if (line.toLowerCase().startsWith("formato ") || line.toLowerCase().startsWith("mensagem ") || line.toLowerCase().startsWith("segundo contato") || line.toLowerCase().startsWith("roteiro do áudio")) {

          if (currentBlock) blocks.push(processBlock(currentBlock));

          currentBlock = { title: line, text: "" };

        } else {

          if (!currentBlock) {

            currentBlock = { title: "Introdução", text: line };

          } else {

            currentBlock.text += (currentBlock.text ? "\n\n" : "") + line;

          }

        }

      });

      if (currentBlock) blocks.push(processBlock(currentBlock));

      approachesData = blocks;

    }


    function processBlock(block) {

      const parsed = extractCopyableAndNotes(block.text);

      return {

        title: block.title,

        text: block.text,

        copyableText: parsed.copyableText,

        notes: parsed.notes

      };

    }

    function formatWhatsLink(raw) {

      if (!raw) return "";

      const trimmed = String(raw).trim();

      if (trimmed.startsWith("http://") || trimmed.startsWith("https://")) return trimmed;

      const digits = trimmed.replace(/\D/g, "");

      if (!digits) return "";

      return `https://wa.me/${digits}`;

    }


    function formatInstaLink(raw) {

      if (!raw) return "";

      const trimmed = String(raw).trim();

      if (trimmed.startsWith("http://") || trimmed.startsWith("https://")) return trimmed;

      const handle = trimmed.replace(/^@/, "");

      return `https://instagram.com/${handle}`;

    }

    function copyText(btn, index) {

      const text = approachesData[index].copyableText;

      

      const textArea = document.createElement("textarea");

      textArea.value = text;

      

      textArea.style.position = "absolute";

      textArea.style.left = "-9999px";

      document.body.appendChild(textArea);

      

      textArea.select();

      

      try {

        document.execCommand('copy');

        const originalText = btn.innerText;

        btn.innerText = "Copiado s/ parênteses!";

        btn.classList.add("bg-emerald-600", "text-white");

        btn.classList.remove("bg-neutral-100", "text-neutral-800");

        

        setTimeout(() => {

          btn.innerText = originalText;

          btn.classList.remove("bg-emerald-600", "text-white");

          btn.classList.add("bg-neutral-100", "text-neutral-800");

        }, 1800);

      } catch (err) {

        console.error("Oops, unable to copy", err);

        showToast("Falha ao copiar texto.", true);

      }

      

      document.body.removeChild(textArea);

    async function saveLead(event) {

      event.preventDefault();

      const idxValue = document.getElementById("leadRowIndex").value;

      const isEdit = idxValue !== "";


      const leadObj = {

        "Data de Entrada": document.getElementById("fieldDataEntrada").value.trim(),

        "Nome do Lead": document.getElementById("fieldNome").value.trim(),

        "WhatsApp": document.getElementById("fieldWhats").value.trim(),

        "Instagram": document.getElementById("fieldInsta").value.trim(),

        "Fase": document.getElementById("fieldFase").value.trim(),

        "Data da Call": document.getElementById("fieldDataCall").value.trim(),

        "Status / Desfecho": document.getElementById("fieldStatus").value.trim(),

        "Valor": document.getElementById("fieldValor").value.trim()

      };


      let sheetRowNumber = null;

      if (isEdit) {

        const existing = crmData[Number(idxValue)];

        sheetRowNumber = existing._rowIndex || (Number(idxValue) + 2);

        crmData[Number(idxValue)] = { ...existing, ...leadObj, _rowIndex: sheetRowNumber };

      } else {

        sheetRowNumber = crmData.length + 2;

        crmData.unshift({ ...leadObj, _rowIndex: sheetRowNumber });

      }


      renderCrm();

      closeLeadModal();


      // Sincroniza com a planilha se a URL do Google Apps Script estiver configurada

      if (GOOGLE_SCRIPT_URL && GOOGLE_SCRIPT_URL.startsWith("https://script.google.com")) {

        isSaving = true;

        showToast("Salvando alterações na planilha...");

        try {

          await fetch(GOOGLE_SCRIPT_URL, {

            method: "POST",

            mode: "no-cors",

            headers: { "Content-Type": "text/plain;charset=utf-8" },

            body: JSON.stringify({

              action: isEdit ? "edit" : "add",

              rowIndex: sheetRowNumber,

              lead: leadObj

            })

          });

          showToast(isEdit ? "Lead atualizado na planilha!" : "Novo lead adicionado na planilha!");

          setTimeout(() => {

            isSaving = false;

            loadAllData(true);

          }, 1500);

        } catch (err) {

          console.error(err);

          isSaving = false;

          showToast("Salvo localmente, mas houve erro ao enviar para a planilha.", true);

        }

      } else {

        showToast("Atualizado na tela! Configure a Sincronização (ícone de engrenagem) para gravar na planilha.");

      }

    }
